    /* the "less often" route — no discount, a better suggestion instead */
    if(f.off === null){
      resultEl.classList.add('is-out');
      titleEl.textContent = v.label;
      priceEl.textContent = 'Not a plan fit';
      metaEl.hidden = true;
      noteEl.hidden = true;
      outEl.hidden = false;
      outEl.innerHTML =
        'Left longer than six weeks, a Maintenance Wash won’t lift what the carpets and seats pick up.' +
        '<br><br><strong style="color:#fff">Better:</strong> an ' +
        '<a href="https://mobilepitstop.uk/interior-deep-clean">Interior Deep Clean</a> or an ' +
        '<a href="https://mobilepitstop.uk/in-out-deep-clean">In &amp; Out Deep Clean</a> whenever the car needs it. No plan required.';
      return;
    }

    resultEl.classList.remove('is-out');
    metaEl.hidden = false;
    noteEl.hidden = false;
    outEl.hidden = true;

    var per = v.price * (1 - f.off / 100);

    titleEl.textContent = v.label + ' • ' + f.label;
    priceEl.innerHTML   = '<span class="was">' + money(v.price) + '</span>' + money(per);
    baseEl.textContent  = money(v.price);
    offEl.textContent   = '−' + f.off + '%  (' + money(v.price - per) + ' a visit)';
    labelEl.textContent = 'Roughly per month (' +
      (f.perMonth === 4.3 ? '4-5' : f.perMonth) + ' visit' + (f.perMonth === 1 ? '' : 's') + ')';
    totalEl.textContent = money(per * f.perMonth);
  }

  sizeEl.addEventListener('change', update);
  freqEl.addEventListener('change', update);
  update();
})();
</script>
