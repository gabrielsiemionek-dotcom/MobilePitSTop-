<script>
(function(){
  /* CODES — the percentage MUST match the Acuity coupon of the same name.
     `expires` is YYYY-MM-DD; the code stops working at the end of that day. */
  var PROMO_CODES = {
    'AUTUMN20': { off:20, label:'Autumn offer', expires:null }
  };

  var PFX = 'fvp';
  var box, flag, title, copy, form, msg;

  function grabEls(){
    box   = document.getElementById(PFX + 'Promo');
    flag  = document.getElementById(PFX + 'PromoFlag');
    title = document.getElementById(PFX + 'PromoTitle');
    copy  = document.getElementById(PFX + 'PromoCopy');
    form  = document.getElementById(PFX + 'PromoForm');
    msg   = document.getElementById(PFX + 'PromoMsg');
    return !!box;
  }

  var TOTAL_IDS = ["fvpTotal", "fvpFinalTotal"];
  var active = null;
  var obs    = null;
  var busy   = false;

  window.MPS_PROMO = window.MPS_PROMO || {};
  window.MPS_PROMO.off  = function(){ return active ? active.off  : 0; };
  window.MPS_PROMO.code = function(){ return active ? active.code : null; };

  function announce(){
    try{ document.dispatchEvent(new CustomEvent('mps:promo')); }catch(e){}
  }

  window.MPS_PROMO.decorate = function(url){
    if(!active || !url) return url;
    if(url.indexOf('certificate=') > -1) return url;
    return url + (url.indexOf('?') > -1 ? '&' : '?') +
           'certificate=' + encodeURIComponent(active.code);
  };

  /* O/0, I/1, S/5 read the same on a phone keyboard, so both sides are
     normalised before comparing and OUR spelling goes to Acuity. */
  function normaliseCode(raw){
    return String(raw || '')
      .toUpperCase()
      .replace(/[^A-Z0-9]/g, '')
      .replace(/O/g, '0')
      .replace(/[IL]/g, '1')
      .replace(/S/g, '5');
  }

  function lookupCode(raw, table){
    var want = normaliseCode(raw);
    if(!want) return null;
    for(var key in table){
      if(Object.prototype.hasOwnProperty.call(table, key) && normaliseCode(key) === want){
        return { code: key, entry: table[key] };
      }
    }
    return null;
  }

  function notExpired(entry){
    if(!entry.expires) return true;
    var end = new Date(entry.expires + 'T23:59:59');
    return !isNaN(end.getTime()) && new Date() <= end;
  }

  function money(n){
    return '\u00a3' + (Math.round(n * 100) / 100).toFixed(2).replace(/\.00$/, '');
  }

  function paintTotals(){
    if(!active) return;
    busy = true;
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(!el) return;
      var raw = el.getAttribute('data-mps-full');
      if(raw === null) raw = el.textContent;
      var m = String(raw).match(/£\s*([\d,]+(?:\.\d+)?)/);
      if(!m) return;
      var full = parseFloat(m[1].replace(/,/g, ''));
      if(!isFinite(full) || full <= 0) return;
      el.setAttribute('data-mps-full', money(full));
      var desired = '<span class="mps-was">' + money(full) + '</span>' +
                    money(full * (1 - active.off / 100));
      if(el.innerHTML !== desired) el.innerHTML = desired;
    });
    if(obs) obs.takeRecords();
    busy = false;
  }

  function clearTotals(){
    busy = true;
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(!el) return;
      var raw = el.getAttribute('data-mps-full');
      if(raw !== null && el.textContent !== raw){ el.textContent = raw; }
      el.removeAttribute('data-mps-full');
    });
    if(obs) obs.takeRecords();
    busy = false;
  }

  function attachObserver(){
    if(!window.MutationObserver) return;
    obs = new MutationObserver(function(){
      if(busy || !active) return;
      TOTAL_IDS.forEach(function(id){
        var el = document.getElementById(id);
        if(el && el.querySelector('.mps-was') === null) el.removeAttribute('data-mps-full');
      });
      paintTotals();
    });
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(el) obs.observe(el, { childList:true, characterData:true, subtree:true });
    });
  }

  function showMsg(text, kind){
    if(!msg) return;
    if(!text){ msg.hidden = true; msg.textContent = ''; return; }
    msg.hidden = false;
    msg.textContent = text;
    msg.className = 'mps-promo-msg ' + (kind || '');
  }

  function renderOn(){
    box.classList.add('is-on');
    flag.textContent  = active.off + '% off applied';
    title.textContent = 'Discount applied';
    copy.hidden = false;
    copy.innerHTML = 'Code <strong>' + active.code + '</strong> is applied &mdash; ' +
      active.off + '% off this booking. It carries through to checkout automatically, ' +
      'so there&rsquo;s no need to type it again.';
    form.innerHTML = '<button class="mps-promo-clear" id="' + PFX + 'PromoClear" type="button">Remove code</button>';
    document.getElementById(PFX + 'PromoClear')
      .addEventListener('click', function(){ clearCode(); });
  }

  function renderOff(){
    box.classList.remove('is-on');
    flag.textContent  = 'Discount code';
    title.textContent = 'Got a code? Enter it here';
    copy.innerHTML = '';
    copy.hidden = true;
    form.innerHTML =
      '<label for="' + PFX + 'PromoInput" style="position:absolute;width:1px;height:1px;overflow:hidden;clip:rect(0,0,0,0)">Discount code</label>' +
      '<input class="mps-promo-input" id="' + PFX + 'PromoInput" type="text" autocomplete="off" spellcheck="false" placeholder="Enter discount code">' +
      '<button class="mps-promo-btn" id="' + PFX + 'PromoApply" type="button">Apply</button>';
    wire();
  }

  function applyCode(raw){
    if(!String(raw || '').trim()){ showMsg('Enter your code first.', 'err'); return; }
    var found = lookupCode(raw, PROMO_CODES);
    if(!found){
      showMsg('That code isn\u2019t recognised. Check it and try again, or give us a ring.', 'err');
      return;
    }
    if(!notExpired(found.entry)){
      showMsg('That code has expired. Get in touch and we\u2019ll see what we can do.', 'err');
      return;
    }
    active = { code: found.code, off: found.entry.off };
    try{ sessionStorage.setItem('mps_promo_code', found.code); }catch(e){}
    renderOn();
    showMsg('');
    paintTotals();
    announce();
  }

  function clearCode(){
    active = null;
    try{ sessionStorage.removeItem('mps_promo_code'); }catch(e){}
    clearTotals();
    renderOff();
    showMsg('');
    announce();
  }

  function wire(){
    var input = document.getElementById(PFX + 'PromoInput');
    var apply = document.getElementById(PFX + 'PromoApply');
    if(apply) apply.addEventListener('click', function(){ applyCode(input ? input.value : ''); });
    if(input){
      input.addEventListener('keydown', function(e){
        if(e.key === 'Enter'){ e.preventDefault(); applyCode(input.value); }
      });
      input.addEventListener('input', function(){ showMsg(''); });
    }
  }

  function start(){
    if(!grabEls()) return;
    renderOff();
    attachObserver();
    var pre = null;
    try{ pre = new URLSearchParams(window.location.search).get('code'); }catch(e){}
    if(!pre){ try{ pre = sessionStorage.getItem('mps_promo_code'); }catch(e){} }
    if(pre && lookupCode(pre, PROMO_CODES)) applyCode(pre);
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', start);
  } else {
    start();
  }
})();
</script>

  function headerOffset(){
    var tallest = 0;
    var nodes = document.querySelectorAll('header, .Header, #header, .sqs-announcement-bar-dropzone');
    for(var i = 0; i < nodes.length; i++){
      var cs = window.getComputedStyle(nodes[i]);
      if(cs.position !== 'fixed' && cs.position !== 'sticky') continue;
      var r = nodes[i].getBoundingClientRect();
      if(r.top <= 4 && r.height > 0) tallest = Math.max(tallest, r.height);
    }
    return tallest ? tallest + 12 : 16;
  }

  var MOTION_OK = !(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);

  /* only scrolls if the question has gone completely out of view */
  function scrollToWidgetTop(){
    var anchor = root.querySelector('.wizard') || root;
    var off = headerOffset();
    var r   = anchor.getBoundingClientRect();
    var vh  = window.innerHeight || document.documentElement.clientHeight;
    if(!(r.bottom < off + 40) && !(r.top > vh - 80)) return;
    var y = Math.max(0, r.top + window.pageYOffset - off - 10);
    if(Math.abs(window.pageYOffset - y) < 10) return;
    window.scrollTo({ top:y, behavior: MOTION_OK ? 'smooth' : 'auto' });
  }

  function loadDays(month){
    var type = bookingTypeId();
    if(!type) return;
    booking.month = month;

    var grid = root.querySelector('#fvpBkDays');
    grid.innerHTML = '';
    root.querySelector('#fvpBkTimesWrap').hidden = true;
    root.querySelector('#fvpBkDaysLoad').hidden = false;
    bkErr.hidden = true;

    var y = +month.slice(0,4), m = +month.slice(5,7);
    root.querySelector('#fvpBkMonthName').textContent = MONTHS[m-1] + ' ' + y;
    /* can't go back past this month */
    var now = new Date();
    root.querySelector('#fvpBkPrev').disabled =
      (y < now.getFullYear()) || (y === now.getFullYear() && m <= now.getMonth() + 1);

           empty grid and making them work it out */

          booking.autoSkipped = true;

      root.querySelector('#fvpBkDaysLoad').hidden = true;
      bkFail('Couldn\u2019t reach the diary. Try again, or message us on WhatsApp.');

  function drawMonth(y, m, free){
    var grid = root.querySelector('#fvpBkDays');
    grid.innerHTML = '';

    var first = new Date(y, m - 1, 1);
    var lead  = (first.getDay() + 6) % 7;          /* weeks start Monday */
    var count = new Date(y, m, 0).getDate();

    for(var i = 0; i < lead; i++){
      var blank = document.createElement('span');
      blank.className = 'bk-cell is-empty';
      grid.appendChild(blank);
    }

    for(var day = 1; day <= count; day++){
      var iso = y + '-' + ('0' + m).slice(-2) + '-' + ('0' + day).slice(-2);
      var b = document.createElement('button');
      b.type = 'button'; b.className = 'bk-cell'; b.textContent = day;
      b.setAttribute('aria-pressed','false');
      if(!free[iso]){
        b.disabled = true;
        b.setAttribute('aria-label', day + ' \u2014 fully booked');
      } else {
        b.setAttribute('data-iso', iso);
        (function(isoDay, btn){
          btn.addEventListener('click', function(){
            Array.prototype.forEach.call(grid.children, function(o){
              if(o.setAttribute) o.setAttribute('aria-pressed','false');
            });
            btn.setAttribute('aria-pressed','true');
            booking.day = isoDay; booking.time = null;
            bkPay.disabled = true; bkPay.textContent = 'Pay the deposit & book';
            root.querySelector('#fvpBkChosen').hidden = true;
            loadTimes(isoDay);
          });
        })(iso, b);
      }
      grid.appendChild(b);
    }
  }

  var slotsByHour = {};

  function loadTimes(iso){
    var type = bookingTypeId();
    root.querySelector('#fvpBkTimesWrap').hidden = false;
    root.querySelector('#fvpBkTimesLoad').hidden = false;
    root.querySelector('#fvpBkHours').innerHTML = '';
    root.querySelector('#fvpBkMins').innerHTML  = '';

    var p = addonParams(); p.type = type; p.when = iso;
    askAcuity(p).then(function(times){
      root.querySelector('#fvpBkTimesLoad').hidden = true;
      slotsByHour = {};
      (times || []).forEach(function(t){
        var raw = t.time || t;
        var hm  = String(raw).match(/T(\d{2}):(\d{2})/);
        if(!hm) return;
        (slotsByHour[hm[1]] = slotsByHour[hm[1]] || []).push({ mm:hm[2], raw:raw });
      });

      var hours = Object.keys(slotsByHour).sort();
      if(!hours.length){
        root.querySelector('#fvpBkHours').innerHTML = '<p class="bk-load" style="padding:10px;">That day filled up.</p>';
        return;
      }
      hours.forEach(function(hh){
        var b = document.createElement('button');
        b.type = 'button'; b.className = 'bk-tick'; b.textContent = hh;
        b.setAttribute('aria-pressed','false');
        b.addEventListener('click', function(){
          Array.prototype.forEach.call(root.querySelector('#fvpBkHours').children, function(o){
            if(o.setAttribute) o.setAttribute('aria-pressed','false');
          });
          b.setAttribute('aria-pressed','true');
          drawMinutes(hh);
        });
        root.querySelector('#fvpBkHours').appendChild(b);
      });
      drawMinutes(hours[0]);
      root.querySelector('#fvpBkHours').firstChild.setAttribute('aria-pressed','true');
    }).catch(function(){
      root.querySelector('#fvpBkTimesLoad').hidden = true;
      bkFail('Couldn\u2019t load that day. Try another.');
    });
  }

    });
  }

    try{ want = new URLSearchParams(window.location.search).get('size'); }catch(e){ return; }
    if(!want) return;
    var value = MAP[String(want).trim().toLowerCase()];
    if(!value) return;
    var el = root.querySelector('.opt[data-pick="vehicle"][data-value="' + value + '"]');
    if(el) setTimeout(function(){ el.click(); }, 0);
  })();

  /* land at the top unless the visitor has already moved or there's a #hash */
  var _touched = false;
  ['touchstart','wheel','keydown','pointerdown','mousedown'].forEach(function(ev){
    window.addEventListener(ev, function(){ _touched = true; }, { once:true, passive:true });
  });
  function settleAtTop(){
    if(_touched || window.location.hash || window.pageYOffset <= 4) return;
    window.scrollTo(0, 0);
  }
  window.addEventListener('load', function(){ syncViewportHeight(); settleAtTop(); });
  setTimeout(function(){ syncViewportHeight(); settleAtTop(); }, 250);
  setTimeout(settleAtTop, 700);
})();
