        }, {passive:true});
        frame.addEventListener('touchmove', function(e){
          if(!dragging || !e.touches[0]) return;
          var dx = e.touches[0].clientX - tX, dy = e.touches[0].clientY - tY;
          if(axis === null){
            if(Math.abs(dx) < AXIS_THRESHOLD && Math.abs(dy) < AXIS_THRESHOLD) return;
            axis = Math.abs(dx) > Math.abs(dy) ? 'x' : 'y';
          }
          if(axis === 'y'){ dragging = false; return; }
          e.preventDefault();
          var r = frame.getBoundingClientRect();
          var p = Math.max(0, Math.min(100, ((e.touches[0].clientX - r.left) / r.width) * 100));
          range.value = Math.round(p); setWidth(p);
        }, {passive:false});
        frame.addEventListener('touchend',    function(){ dragging = false; axis = null; });
        frame.addEventListener('touchcancel', function(){ dragging = false; axis = null; });
      }
    }

    /* -- CTA glow -- */
    var cta = document.getElementById('mvCTA');
    if(cta && 'IntersectionObserver' in window){
      new IntersectionObserver(function(entries){
        entries.forEach(function(e){ cta.classList.toggle('is-visible', e.isIntersecting); });
      }, {threshold:0.5}).observe(cta);
    }

    /* -- Carousels -- */
    function initCarousel(gridId, prevId, nextId, dotsId, label){
      var grid = document.getElementById(gridId);
      var prev = document.getElementById(prevId);
      var next = document.getElementById(nextId);
      var dots = document.getElementById(dotsId);
      if(!grid || !prev || !next || !dots) return;

      var cards = Array.prototype.slice.call(grid.querySelectorAll('.mps-vcard'));
      if(!cards.length){ prev.parentNode.style.display = 'none'; return; }
      var idx = 0;

      function perView(){
        var w = cards[0].getBoundingClientRect().width;
        if(!w) return 1;
        return Math.max(1, Math.round(grid.clientWidth / w));
      }
      function maxIdx(){ return Math.max(0, cards.length - perView()); }

      cards.forEach(function(_, i){
        var b = document.createElement('button');
        b.type = 'button';
        b.setAttribute('aria-label', label + ' ' + (i + 1));
        b.addEventListener('click', function(){ goTo(i); });
        dots.appendChild(b);
      });
      var dotEls = Array.prototype.slice.call(dots.children);

      function paint(){
        var mx = maxIdx();
        dotEls.forEach(function(d, i){
          d.classList.toggle('is-active', i === idx);
          d.style.display = (i > mx) ? 'none' : '';
        });
        prev.disabled = (idx <= 0);
        next.disabled = (idx >= mx);
        prev.parentNode.style.visibility = (mx === 0) ? 'hidden' : '';
      }
      function goTo(i){
        idx = Math.max(0, Math.min(maxIdx(), i));
        var card = cards[idx];
        if(card) grid.scrollTo({ left: card.offsetLeft - grid.offsetLeft, behavior:'smooth' });
        paint();
      }

      var rt = null;
      window.addEventListener('resize', function(){
        clearTimeout(rt);
        rt = setTimeout(function(){ if(idx > maxIdx()) goTo(maxIdx()); else paint(); }, 150);
      });
      prev.addEventListener('click', function(){ goTo(idx - 1); });
      next.addEventListener('click', function(){ goTo(idx + 1); });

      var t = null;
      grid.addEventListener('scroll', function(){
        clearTimeout(t);
        t = setTimeout(function(){
          var nearest = 0, best = Infinity;
          cards.forEach(function(c, i){
            var d = Math.abs((c.offsetLeft - grid.offsetLeft) - grid.scrollLeft);
            if(d < best){ best = d; nearest = i; }
          });
          nearest = Math.min(nearest, maxIdx());
          if(nearest !== idx){ idx = nearest; paint(); }
        }, 80);
      }, { passive:true });

      paint();
    }
    initCarousel('mvVisualGrid', 'mvCarPrev',   'mvCarNext',   'mvCarDots',   'Go to card');
    initCarousel('mvAddonGrid',  'mvAddonPrev', 'mvAddonNext', 'mvAddonDots', 'Go to add-on');

    /* -- Add-on cards open their matching accordion -- */
    Array.prototype.slice.call(document.querySelectorAll('#learnMoreMiniValet [data-open-addon]'))
      .forEach(function(btn){
        btn.addEventListener('click', function(){
          var target = document.getElementById(btn.getAttribute('data-open-addon'));
          if(!target) return;
          Array.prototype.slice.call(document.querySelectorAll(
            '#learnMoreMiniValet .mps-acc details[open], #mvFAQs details[open]'
          )).forEach(function(d){ if(d !== target) d.open = false; });
          target.open = true;
          target.scrollIntoView({ behavior:'smooth', block:'center' });
          target.classList.add('mps-flash');
          setTimeout(function(){ target.classList.remove('mps-flash'); }, 1600);
        });
      });

    /* -- Accordions: only one open at a time -- */
    var allSummaries = Array.prototype.slice.call(document.querySelectorAll(
      '#learnMoreMiniValet .mps-acc details summary, #mvFAQs details summary'
    ));
    allSummaries.forEach(function(summary){
      summary.addEventListener('click', function(){
        var thisDetails = summary.parentElement;
        allSummaries.forEach(function(other){
          var od = other.parentElement;
          if(od !== thisDetails && od.open) od.open = false;
        });
      });
    });
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', init);
  } else {
    setTimeout(init, 0);
  }

})();
</script>
