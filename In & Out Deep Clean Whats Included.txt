<script>
(function(){

  function init(){


    /* -- Before/After slider -------------------------------------- */
    var afterWrap = document.getElementById('fvAfterWrap');
    var range     = document.getElementById('fvCompareRange');
    var frame     = document.querySelector('.mps-compare .frame');

    if(afterWrap && range){
      var afterImg = afterWrap.querySelector('img');

      function sizeAfterImg(){
        if(frame && afterImg){
          afterImg.style.width = frame.offsetWidth + 'px';
        }
      }
      sizeAfterImg();
      window.addEventListener('resize', sizeAfterImg);

      function setWidth(v){
        afterWrap.style.width = v + '%';
      }
      setWidth(range.value);
      range.addEventListener('input',  function(){ setWidth(this.value); });
      range.addEventListener('change', function(){ setWidth(this.value); });

      if(frame){
        var dragging = false;
        frame.addEventListener('mousedown',  function(){ dragging = true; });
        frame.addEventListener('mousemove',  function(e){
          if(!dragging) return;
          var r   = frame.getBoundingClientRect();
          var pct = Math.max(0, Math.min(100, ((e.clientX - r.left) / r.width) * 100));
          range.value = Math.round(pct);
          setWidth(pct);
        });
        frame.addEventListener('mouseup',    function(){ dragging = false; });
        frame.addEventListener('mouseleave', function(){ dragging = false; });

        /* ── Touch handling ────────────────────────────────────────
           The old version called preventDefault() on touchstart, which
           cancels the gesture before the browser has decided whether it's
           a scroll or a drag. Because this image is full-width on a phone,
           anyone starting a swipe on it simply couldn't scroll the page.

           Now we wait for the first few pixels of movement, work out which
           way the finger is going, and only claim the gesture if it's
           clearly horizontal. A vertical swipe is handed back to the page.
           ──────────────────────────────────────────────────────────── */
        var tX = 0, tY = 0, axis = null;
        var AXIS_THRESHOLD = 8;   /* px of travel before we decide */

        frame.addEventListener('touchstart', function(e){
          if(!e.touches[0]) return;
          tX = e.touches[0].clientX;
          tY = e.touches[0].clientY;
          axis = null;
          dragging = true;
        }, {passive:true});

        frame.addEventListener('touchmove', function(e){
          if(!dragging || !e.touches[0]) return;

          var dx = e.touches[0].clientX - tX;
          var dy = e.touches[0].clientY - tY;

          if(axis === null){
            if(Math.abs(dx) < AXIS_THRESHOLD && Math.abs(dy) < AXIS_THRESHOLD) return;
            axis = Math.abs(dx) > Math.abs(dy) ? 'x' : 'y';
          }

          /* vertical swipe — let the page scroll, and stay out of the way */
          if(axis === 'y'){ dragging = false; return; }

          e.preventDefault();
          var r   = frame.getBoundingClientRect();
          var pct = Math.max(0, Math.min(100, ((e.touches[0].clientX - r.left) / r.width) * 100));
          range.value = Math.round(pct);
          setWidth(pct);
        }, {passive:false});

        frame.addEventListener('touchend',    function(){ dragging = false; axis = null; });
        frame.addEventListener('touchcancel', function(){ dragging = false; axis = null; });
        frame.style.cursor = 'col-resize';
      }
    }

    /* -- Review count from the sitewide MPS_REVIEWS, if set -------- */
    var trustRv = document.getElementById('fvTrustReviews');
    if(trustRv && window.MPS_REVIEWS && window.MPS_REVIEWS.count){
      trustRv.textContent = Number(window.MPS_REVIEWS.count) + ' Google reviews';
    }

    /* -- CTA glow - only when visible ------------------------------ */
    var cta = document.getElementById('fvCTA');
    if(cta && 'IntersectionObserver' in window){
      new IntersectionObserver(function(entries){
        entries.forEach(function(e){
          cta.classList.toggle('is-visible', e.isIntersecting);
        });
      }, {threshold:0.5}).observe(cta);
    }

    /* -- Carousels: reusable, one per grid ------------------------
       Only active visually below 760px (CSS turns the grid into a
       horizontal snap slider); on desktop both grids are 3-col rows. */
    function initCarousel(gridId, prevId, nextId, dotsId, label){
      var grid = document.getElementById(gridId);
      var prev = document.getElementById(prevId);
      var next = document.getElementById(nextId);
      var dots = document.getElementById(dotsId);
      if(!grid || !prev || !next || !dots) return;

      var cards = Array.prototype.slice.call(grid.querySelectorAll('.mps-vcard'));
      if(!cards.length) return;
      var idx = 0;

      /* how many cards fit in view — 1 on mobile, 3 in the desktop
         add-ons slider. Measured rather than assumed so it stays right
         if the card count or breakpoint changes.                      */
      function perView(){
        var w = cards[0].getBoundingClientRect().width;
        if(!w) return 1;
        return Math.max(1, Math.round(grid.clientWidth / w));
      }
      function maxIdx(){
        return Math.max(0, cards.length - perView());
      }

      cards.forEach(function(_, i){
        var b = document.createElement('button');
        b.type = 'button';
        b.setAttribute('aria-label', label + ' ' + (i + 1));
        b.addEventListener('click', function(){ goTo(i); });
        dots.appendChild(b);
      });
      var dotEls = Array.prototype.slice.call(dots.children);

      function paint(){
        var mx = maxIdx();
        dotEls.forEach(function(d, i){
          d.classList.toggle('is-active', i === idx);
          /* hide dots that can never be a start position */
          d.style.display = (i > mx) ? 'none' : '';
        });
        prev.disabled = (idx <= 0);
        next.disabled = (idx >= mx);
      }

      function goTo(i){
        idx = Math.max(0, Math.min(maxIdx(), i));
        var card = cards[idx];
        if(card){
          grid.scrollTo({ left: card.offsetLeft - grid.offsetLeft, behavior: 'smooth' });
        }
        paint();
      }

      prev.addEventListener('click', function(){ goTo(idx - 1); });
      next.addEventListener('click', function(){ goTo(idx + 1); });

      var rt = null;
      window.addEventListener('resize', function(){
        clearTimeout(rt);
        rt = setTimeout(function(){
          if(idx > maxIdx()) goTo(maxIdx()); else paint();
        }, 150);
      });

      var t = null;
      grid.addEventListener('scroll', function(){
        clearTimeout(t);
        t = setTimeout(function(){
          var nearest = 0, best = Infinity;
          cards.forEach(function(c, i){
            var d = Math.abs((c.offsetLeft - grid.offsetLeft) - grid.scrollLeft);
            if(d < best){ best = d; nearest = i; }
          });
          nearest = Math.min(nearest, maxIdx());
          if(nearest !== idx){ idx = nearest; paint(); }
        }, 80);
      }, { passive: true });

      paint();
    }

    initCarousel('fvVisualGrid', 'fvCarPrev', 'fvCarNext', 'fvCarDots', 'Go to card');
    initCarousel('fvAddonGrid',  'fvAddonPrev', 'fvAddonNext', 'fvAddonDots', 'Go to upgrade');

    /* -- Card buttons open the matching add-on accordion ---------- */
    Array.prototype.slice.call(document.querySelectorAll('[data-open-addon]'))
      .forEach(function(btn){
        btn.addEventListener('click', function(){
          var target = document.getElementById(btn.getAttribute('data-open-addon'));
          if(!target) return;

          /* close any other open accordion first, same as clicking one */
          Array.prototype.slice.call(document.querySelectorAll(
            '#WhatIncludedInAndOut .mps-acc details[open], #fvFAQs details[open]'
          )).forEach(function(d){ if(d !== target) d.open = false; });

          target.open = true;
          target.scrollIntoView({ behavior:'smooth', block:'center' });

          target.classList.add('mps-flash');
          setTimeout(function(){ target.classList.remove('mps-flash'); }, 1600);
        });
      });

    /* -- Accordions: only one open at a time -----------------------
       BUGFIX: this previously targeted the booking wizard, which is
       the booking wizard — these accordions live in the section below
       it, so the close-others behaviour never ran on them.           */
    var allSummaries = Array.prototype.slice.call(document.querySelectorAll(
      '#WhatIncludedInAndOut .mps-acc details summary, #fvFAQs details summary'
    ));

    allSummaries.forEach(function(summary){
      summary.addEventListener('click', function(){
        var thisDetails = summary.parentElement;
        allSummaries.forEach(function(other){
          var otherDetails = other.parentElement;
          if(otherDetails !== thisDetails && otherDetails.open){
            otherDetails.open = false;
          }
        });
      });
    });
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', init);
  } else {
    setTimeout(init, 0);
  }

})();
</script>
