  }

  function money(n){
    return '\u00a3' + (Math.round(n * 100) / 100).toFixed(2).replace(/\.00$/, '');
  }

  function paintTotals(){
    if(!active) return;
    busy = true;
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(!el) return;
      var raw = el.getAttribute('data-mps-full');
      if(raw === null) raw = el.textContent;
      var m = String(raw).match(/£\s*([\d,]+(?:\.\d+)?)/);
      if(!m) return;
      var full = parseFloat(m[1].replace(/,/g, ''));
      if(!isFinite(full) || full <= 0) return;
      el.setAttribute('data-mps-full', money(full));
      var desired = '<span class="mps-was">' + money(full) + '</span>' +
                    money(full * (1 - active.off / 100));
      if(el.innerHTML !== desired) el.innerHTML = desired;
    });
    if(obs) obs.takeRecords();
    busy = false;
  }

  function clearTotals(){
    busy = true;
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(!el) return;
      var raw = el.getAttribute('data-mps-full');
      if(raw !== null && el.textContent !== raw){ el.textContent = raw; }
      el.removeAttribute('data-mps-full');
    });
    if(obs) obs.takeRecords();
    busy = false;
  }

  function attachObserver(){
    if(!window.MutationObserver) return;
    obs = new MutationObserver(function(){
      if(busy || !active) return;
      TOTAL_IDS.forEach(function(id){
        var el = document.getElementById(id);
        if(el && el.querySelector('.mps-was') === null) el.removeAttribute('data-mps-full');
      });
      paintTotals();
    });
    TOTAL_IDS.forEach(function(id){
      var el = document.getElementById(id);
      if(el) obs.observe(el, { childList:true, characterData:true, subtree:true });
    });
  }

  function showMsg(text, kind){
    if(!msg) return;
    if(!text){ msg.hidden = true; msg.textContent = ''; return; }
    msg.hidden = false;
    msg.textContent = text;
    msg.className = 'mps-promo-msg ' + (kind || '');
  }

  function renderOn(){
    box.classList.add('is-on');
    flag.textContent  = active.off + '% off applied';
    title.textContent = 'Discount applied';
    copy.hidden = false;
    copy.innerHTML = 'Code <strong>' + active.code + '</strong> is applied &mdash; ' +
      active.off + '% off this booking. It carries through to checkout automatically, ' +
      'so there&rsquo;s no need to type it again.';
    form.innerHTML = '<button class="mps-promo-clear" id="' + PFX + 'PromoClear" type="button">Remove code</button>';
    document.getElementById(PFX + 'PromoClear')
      .addEventListener('click', function(){ clearCode(); });
  }

  function renderOff(){
    box.classList.remove('is-on');
    flag.textContent  = 'Discount code';
    title.textContent = 'Got a code? Enter it here';
    copy.innerHTML = '';
    copy.hidden = true;
    form.innerHTML =
      '<label for="' + PFX + 'PromoInput" style="position:absolute;width:1px;height:1px;overflow:hidden;clip:rect(0,0,0,0)">Discount code</label>' +
      '<input class="mps-promo-input" id="' + PFX + 'PromoInput" type="text" autocomplete="off" spellcheck="false" placeholder="Enter discount code">' +
      '<button class="mps-promo-btn" id="' + PFX + 'PromoApply" type="button">Apply</button>';
    wire();
  }

  function applyCode(raw){
    if(!String(raw || '').trim()){ showMsg('Enter your code first.', 'err'); return; }
    var found = lookupCode(raw, PROMO_CODES);
    if(!found){
      showMsg('That code isn\u2019t recognised. Check it and try again, or give us a ring.', 'err');
      return;
    }
    if(!notExpired(found.entry)){
      showMsg('That code has expired. Get in touch and we\u2019ll see what we can do.', 'err');
      return;
    }
    active = { code: found.code, off: found.entry.off };
    try{ sessionStorage.setItem('mps_promo_code', found.code); }catch(e){}
    renderOn();
    showMsg('');
    paintTotals();
    announce();
  }

  function clearCode(){
    active = null;
    try{ sessionStorage.removeItem('mps_promo_code'); }catch(e){}
    clearTotals();
    renderOff();
    showMsg('');
    announce();
  }

  function wire(){
    var input = document.getElementById(PFX + 'PromoInput');
    var apply = document.getElementById(PFX + 'PromoApply');
    if(apply) apply.addEventListener('click', function(){ applyCode(input ? input.value : ''); });
    if(input){
      input.addEventListener('keydown', function(e){
        if(e.key === 'Enter'){ e.preventDefault(); applyCode(input.value); }
      });
      input.addEventListener('input', function(){ showMsg(''); });
    }
  }

  function start(){
    if(!grabEls()) return;
    renderOff();
    attachObserver();
    var pre = null;
    try{ pre = new URLSearchParams(window.location.search).get('code'); }catch(e){}
    if(!pre){ try{ pre = sessionStorage.getItem('mps_promo_code'); }catch(e){} }
    if(pre && lookupCode(pre, PROMO_CODES)) applyCode(pre);
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', start);
  } else {
    start();
  }
})();
</script>

  function fmtGBP(n){ return '\u00a3' + Number(n || 0).toFixed(0); }

  /* ---- prices ---- */
  function sizeBand(){ return state.vehicle === '7seats' ? 'large' : 'small'; }
  function seatsPrice(){ return (state.seats === 'yes' && state.fabric) ? SEATS[state.fabric].price : 0; }
  function protPrice(){
    if(state.prot !== 'yes' || !state.fabric || !state.vehicle) return 0;
    return PROT[state.fabric][sizeBand()].price;
  }
  function calcTotal(){
    if(!state.vehicle) return 0;
    var t = BASE_PRICES[state.vehicle] || 0;
    t += seatsPrice();
    t += protPrice();
    if(state.odour   === 'yes') t += ADDON_ODOUR.price;
    if(state.sand    === 'yes') t += ADDON_SAND.price;
    if(state.pet     === 'yes') t += ADDON_PET.price;
    if(state.roof    === 'yes') t += ADDON_ROOF.price;
    if(state.antifog === 'yes') t += ADDON_FOG.price;
    return t;
  }
  function promoOff(){
    try{ return (window.MPS_PROMO && window.MPS_PROMO.off) ? (window.MPS_PROMO.off() || 0) : 0; }
    catch(e){ return 0; }
  }
  function netTotal(){ return calcTotal() * (1 - promoOff() / 100); }
  function depositAmount(){ return Math.max(1, Math.round(netTotal() * DEPOSIT_PCT) / 100); }

  function bookingTypeId(){
    return (state.more || !state.vehicle) ? null : (SIZE_TYPE[state.vehicle] || null);
  }
  function addonIds(){
    var a = [];
    if(state.seats === 'yes' && state.fabric) a.push(SEATS[state.fabric].id);
    if(state.prot  === 'yes' && state.fabric) a.push(PROT[state.fabric][sizeBand()].id);
    if(state.odour   === 'yes') a.push(ADDON_ODOUR.id);
    if(state.sand    === 'yes') a.push(ADDON_SAND.id);
    if(state.pet     === 'yes') a.push(ADDON_PET.id);
    if(state.roof    === 'yes') a.push(ADDON_ROOF.id);
    if(state.antifog === 'yes') a.push(ADDON_FOG.id);
    return a.filter(Boolean);
  }

      add += addedRow('seats',

    if(state.prot === 'yes' && state.fabric){

      add += addedRow('prot', PROT_LABELS[state.fabric], protPrice(), 'protection');

    if(!viewportEl) return;
    var active = steps.filter(function(s){
      return Number(s.getAttribute('data-step')) === state.step && !s.hidden;

  function goTo(i){
    state.step = Math.max(0, Math.min(DONE, i));
    track.style.transform = 'translateX(-' + (state.step * 100) + '%)';
    root.classList.toggle('is-done',        state.step === DONE);
    root.classList.toggle('is-first',       state.step === 0);
    root.classList.toggle('is-questioning', state.step > 0);
    btnBack.disabled = state.step === 0;
    renderProg();
    update();
  }
  function advance(){ if(state.step < DONE) goTo(state.step + 1); }

  function clearStep(i){
    steps.forEach(function(s){
      if(Number(s.getAttribute('data-step')) !== i) return;
      s.querySelectorAll('.opt[role="option"]').forEach(function(o){ o.setAttribute('aria-selected','false'); });
    });
  }
  function resetAfter(pick){
    var idx = ORDER.indexOf(pick);
    for(var k = idx + 1; k < ORDER.length; k++){ state[ORDER[k]] = null; clearStep(k); }
  }

  root.querySelectorAll('.opt[role="option"]').forEach(function(opt){
    opt.addEventListener('click', function(){ activate(opt); });
    opt.addEventListener('keydown', function(e){
      if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); activate(opt); }
    });
  });

  function activate(opt){
    var pick = opt.getAttribute('data-pick'), value = opt.getAttribute('data-value');

    if(pick === 'more'){
      state.more = value; state.vehicle = null;
      resetAfter('vehicle'); clearStep(0);
      opt.setAttribute('aria-selected','true');
      update(); goTo(DONE);
      return;
    }
    if(pick === 'vehicle') state.more = null;

    var stepIdx = Number(opt.closest('.step').getAttribute('data-step'));
    resetAfter(pick);
    clearStep(stepIdx);
    state[pick] = value;
    opt.setAttribute('aria-selected','true');
    update();
    advance();
  }

  function monthStr(d){ return d.getFullYear() + '-' + ('0' + (d.getMonth()+1)).slice(-2); }

  function askAcuity(params){
    var q = Object.keys(params).map(function(k){
      return encodeURIComponent(k) + '=' + encodeURIComponent(params[k]);
    }).join('&');
    return fetch(AVAILABILITY + '?token=' + encodeURIComponent(MPS_TOKEN) + '&' + q)
      .then(function(r){ return r.json(); });
  }
  function addonParams(){
    var p = {};
    addonIds().slice(0,4).forEach(function(id,i){ p['a' + (i+1)] = id; });
    return p;
  }

  var MONTHS = ['January','February','March','April','May','June',
                'July','August','September','October','November','December'];

  function loadDays(month){
    var type = bookingTypeId();
    if(!type) return;
    booking.month = month;

    var grid = root.querySelector('#idcBkDays');
    grid.innerHTML = '';
    root.querySelector('#idcBkTimesWrap').hidden = true;
    root.querySelector('#idcBkDaysLoad').hidden = false;
    bkErr.hidden = true;

    var y = +month.slice(0,4), m = +month.slice(5,7);
    root.querySelector('#idcBkMonthName').textContent = MONTHS[m-1] + ' ' + y;
    /* can't go back past this month */
    var now = new Date();
    root.querySelector('#idcBkPrev').disabled =
      (y < now.getFullYear()) || (y === now.getFullYear() && m <= now.getMonth() + 1);

    var p = addonParams(); p.type = type; p.month = month;
    askAcuity(p).then(function(days){
      root.querySelector('#idcBkDaysLoad').hidden = true;
      var free = {};
      (days || []).forEach(function(d){ free[d.date || d] = true; });
      drawMonth(y, m, free);
      if(!days || !days.length){
        /* nothing left this month — step forward rather than showing an
           empty grid and making them work it out */
        if(!booking.autoSkipped){
          booking.autoSkipped = true;
          shiftMonth(1);
          return;
        }
        bkFail('Nothing free in ' + MONTHS[m-1] + ' \u2014 try the next month.');
      }
    }).catch(function(){
      root.querySelector('#idcBkDaysLoad').hidden = true;
      bkFail('Couldn\u2019t reach the diary. Try again, or message us on WhatsApp.');
    });
  }

  function drawMonth(y, m, free){
    var grid = root.querySelector('#idcBkDays');
    grid.innerHTML = '';

    var first = new Date(y, m - 1, 1);
    var lead  = (first.getDay() + 6) % 7;          /* weeks start Monday */
    var count = new Date(y, m, 0).getDate();

    for(var i = 0; i < lead; i++){
      var blank = document.createElement('span');
      blank.className = 'bk-cell is-empty';
      grid.appendChild(blank);
    }

    for(var day = 1; day <= count; day++){
      var iso = y + '-' + ('0' + m).slice(-2) + '-' + ('0' + day).slice(-2);
      var b = document.createElement('button');
      b.type = 'button'; b.className = 'bk-cell'; b.textContent = day;
      b.setAttribute('aria-pressed','false');
      if(!free[iso]){
        b.disabled = true;
        b.setAttribute('aria-label', day + ' \u2014 fully booked');
      } else {
        b.setAttribute('data-iso', iso);
        (function(isoDay, btn){
          btn.addEventListener('click', function(){
            Array.prototype.forEach.call(grid.children, function(o){
              if(o.setAttribute) o.setAttribute('aria-pressed','false');
            });
            btn.setAttribute('aria-pressed','true');
            booking.day = isoDay; booking.time = null;
            bkPay.disabled = true; bkPay.textContent = 'Pay the deposit & book';
            root.querySelector('#idcBkChosen').hidden = true;
            loadTimes(isoDay);
          });
        })(iso, b);
      }
      grid.appendChild(b);
    }
  }

  /* ---- hour and minute wheels, built from what's actually free ---- */
  var slotsByHour = {};

  function loadTimes(iso){
    var type = bookingTypeId();
    root.querySelector('#idcBkTimesWrap').hidden = false;
    root.querySelector('#idcBkTimesLoad').hidden = false;
    root.querySelector('#idcBkHours').innerHTML = '';
    root.querySelector('#idcBkMins').innerHTML  = '';

    var p = addonParams(); p.type = type; p.when = iso;
    askAcuity(p).then(function(times){
      root.querySelector('#idcBkTimesLoad').hidden = true;
      slotsByHour = {};
      (times || []).forEach(function(t){
        var raw = t.time || t;
        var hm  = String(raw).match(/T(\d{2}):(\d{2})/);
        if(!hm) return;
        (slotsByHour[hm[1]] = slotsByHour[hm[1]] || []).push({ mm:hm[2], raw:raw });
      });

      var hours = Object.keys(slotsByHour).sort();
      if(!hours.length){
        root.querySelector('#idcBkHours').innerHTML = '<p class="bk-load" style="padding:10px;">That day filled up.</p>';
        return;
      }
      hours.forEach(function(hh){
        var b = document.createElement('button');
        b.type = 'button'; b.className = 'bk-tick'; b.textContent = hh;
        b.setAttribute('aria-pressed','false');
        b.addEventListener('click', function(){
          Array.prototype.forEach.call(root.querySelector('#idcBkHours').children, function(o){
            if(o.setAttribute) o.setAttribute('aria-pressed','false');
          });
          b.setAttribute('aria-pressed','true');
          drawMinutes(hh);
        });
        root.querySelector('#idcBkHours').appendChild(b);
      });
      drawMinutes(hours[0]);
      root.querySelector('#idcBkHours').firstChild.setAttribute('aria-pressed','true');
    }).catch(function(){
      root.querySelector('#idcBkTimesLoad').hidden = true;
      bkFail('Couldn\u2019t load that day. Try another.');
    });
  }

  function depositAmount(){

  }

  function drawMinutes(hh){
    var wrap = root.querySelector('#idcBkMins');
    wrap.innerHTML = '';
    (slotsByHour[hh] || []).forEach(function(slot){
      var b = document.createElement('button');
      b.type = 'button'; b.className = 'bk-tick'; b.textContent = slot.mm;
      b.setAttribute('aria-pressed','false');
      b.addEventListener('click', function(){
        Array.prototype.forEach.call(wrap.children, function(o){
          if(o.setAttribute) o.setAttribute('aria-pressed','false');
        });
        b.setAttribute('aria-pressed','true');
        booking.time = slot.raw;

        var chosen = root.querySelector('#idcBkChosen');
        chosen.textContent = new Date(String(slot.raw).slice(0,10) + 'T12:00:00')
          .toLocaleDateString('en-GB', { weekday:'long', day:'numeric', month:'long' }) +
          ' at ' + hh + ':' + slot.mm;
        chosen.hidden = false;

        bkDetails.hidden = false;
        bkPay.disabled = false;
        bkPay.textContent = 'Pay ' + fmtGBP(depositAmount()) + ' deposit & book';
        var fine = root.querySelector('#idcBkFine');
        if(fine){
          fine.textContent = fmtGBP(depositAmount()) + ' now holds the slot. The remaining ' +
            fmtGBP(calcTotal() - depositAmount()) + ' is paid on the day, once you\u2019ve seen the work.';
        }
      });
      wrap.appendChild(b);
    });
  }

    if(!last)  return bkFail('And a surname.', root.querySelector('#idcBkLast'));
    if(!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return bkFail('That email doesn\u2019t look right \u2014 your confirmation goes there.', root.querySelector('#idcBkEmail'));
    if(phone.replace(/\D/g,'').length < 10) return bkFail('A mobile number, so we can say when we\u2019re on the way.', root.querySelector('#idcBkPhone'));
    if(addr.length < 8) return bkFail('The full address and postcode \u2014 we\u2019re driving to it.', root.querySelector('#idcBkAddr'));
    if(!car) return bkFail('Make, model and year, so we bring the right kit.', root.querySelector('#idcBkCar'));
    if(booking.answers.agree !== 'yes') return bkFail('Please confirm you\u2019re happy for us to quote before any extra work.');
    if(!booking.answers.power || !booking.answers.water || !booking.answers.space)
      return bkFail('Answer the three yes/no questions and we\u2019ll know what to bring.');

    bkPay.disabled = true;
    bkPay.textContent = 'Taking you to checkout\u2026';

    var label = '10% deposit \u2014 Interior Deep Clean, ' + (VEHICLE_LABELS[state.vehicle] || state.vehicle) +
                (state.fabric ? ', ' + FABRIC_LABELS[state.fabric] : '') +
                (state.seats === 'yes' ? ' + seats' : '') +
                (state.prot  === 'yes' ? ' + protection' : '');

    var p = {
      token: MPS_TOKEN,
      deposit: Math.round(depositAmount() * 100),
      label: label,
      type: bookingTypeId(),
      datetime: booking.time,
      addons: '[' + addonIds().join(',') + ']',
      firstName: first, lastName: last, email: email, phone: phone, address: addr,
      vehicle: car,
      power: booking.answers.power,
      water: booking.answers.water,
      space: booking.answers.space,
      agree: booking.answers.agree,
      marketing: booking.answers.marketing || 'no',
      certificate: (window.MPS_PROMO && window.MPS_PROMO.code) ? (window.MPS_PROMO.code() || '') : '',
      notes: 'Total ' + fmtGBP(netTotal()) + ' \u2014 ' + car,
      source: ''
    };

    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({ event:'pf_book_click', component:'price_finder',
      service:'interior_deep_clean', vehicle_type:state.vehicle, fabric:state.fabric,
      seats:state.seats, protection:state.prot, odour:state.odour,
      price_total:calcTotal(), deposit:depositAmount() });

    var q = Object.keys(p).map(function(k){
      return encodeURIComponent(k) + '=' + encodeURIComponent(p[k]);
    }).join('&');

    fetch(DEPOSIT_HOOK + '?' + q)
      .then(function(r){ return r.json(); })
      .then(function(session){
        if(session && session.url){ window.location.href = session.url; return; }
        throw new Error('no url');
      })
      .catch(function(){
        bkPay.disabled = false;
        bkPay.textContent = 'Pay ' + fmtGBP(depositAmount()) + ' deposit & book';
        bkErr.innerHTML = 'Checkout didn\u2019t open. Nothing has been charged and nothing is booked \u2014 ' +
          '<a href="https://wa.me/447592196929" target="_blank" rel="noopener" style="color:var(--ocean);font-weight:900;">message us</a> ' +
          'and we\u2019ll book it by hand.';
        bkErr.hidden = false;
      });
  });

  /* ---- init ---- */
  goTo(0);
  update();
})();
