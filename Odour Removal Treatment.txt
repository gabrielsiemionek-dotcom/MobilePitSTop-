  function normaliseCode(raw){
    return String(raw || '')
      .toUpperCase()
      .replace(/[^A-Z0-9]/g, '')
      .replace(/O/g, '0')
      .replace(/[IL]/g, '1')
      .replace(/S/g, '5');
  }

  function lookupCode(raw, table){
    var want = normaliseCode(raw);
    if(!want) return null;
    for(var key in table){
      if(Object.prototype.hasOwnProperty.call(table, key) && normaliseCode(key) === want){
        return { code: key, entry: table[key] };
      }
    }
    return null;
  }

    if(!entry.expires) return true;
    var end = new Date(entry.expires + 'T23:59:59');
    return !isNaN(end.getTime()) && new Date() <= end;
  }

    return '\u00a3' + (Math.round(n * 100) / 100).toFixed(2).replace(/\.00$/, '');
  }

    title.textContent = 'Got a code? Enter it here';

      '<button class="mps-promo-btn" id="' + PFX + 'PromoApply" type="button">Apply</button>';
    wire();
  }

    if(!String(raw || '').trim()){ showMsg('Enter your code first.', 'err'); return; }
    var found = lookupCode(raw, PROMO_CODES);
    if(!found){
      showMsg('That code isn\u2019t recognised. Check it and try again, or give us a ring.', 'err');
      return;
    }
    if(!notExpired(found.entry)){
      showMsg('That code has expired. Get in touch and we\u2019ll see what we can do.', 'err');
      return;
    }
    active = { code: found.code, off: found.entry.off };
    try{ sessionStorage.setItem('mps_promo_code', found.code); }catch(e){}
    renderOn();
    showMsg('');
    paintTotals();
    announce();
  }

    active = null;
    try{ sessionStorage.removeItem('mps_promo_code'); }catch(e){}

    renderOff();
    showMsg('');

      });
      input.addEventListener('input', function(){ showMsg(''); });
    }
  }

  }

    document.addEventListener('DOMContentLoaded', start);
  } else {
    start();
  }
})();

  /* ======================================================================
     FUNNEL TRACKING
     dataLayer pushes only. GTM (GTM-MMHW32R2) picks them up and sends
     them to GA4, so there is no gtag/GA code here. Values are numbers in
     pounds, after any promo. Never put a name, email, phone, address,
     postcode or reg into these.
       quote_start     first choice made           (once per page load)
       quote_complete  first full price shown      (once per page load)
       slot_select     a date and time picked
       begin_checkout  Pay deposit pressed, before the Stripe redirect
       generate_lead   a WhatsApp or email request sent
     ====================================================================== */
  var FUNNEL_SERVICE = 'odour_removal';
  var FUNNEL_LABEL   = 'Odour Removal Treatment';
  var funnelSent;
  function funnelPush(o){
    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push(o);
  }
  function funnelOnce(key, o){
    funnelSent = funnelSent || {};
    if(funnelSent[key]) return;
    funnelSent[key] = true;
    funnelPush(o);
  }
  function funnelValue(){ return Math.round((netTotal()) * 100) / 100; }
  function trackQuoteStart(){
    funnelOnce('start', { event:'quote_start', service:FUNNEL_SERVICE });
  }
  /* every required choice made, so the price on screen is the full one */
  function funnelQuoteReady(){
    return !!(state.vehicle && state.smell);
  }
  function trackQuoteComplete(){
    if(!funnelQuoteReady()) return;
    funnelOnce('complete', { event:'quote_complete', service:FUNNEL_SERVICE,
                             value:funnelValue(), currency:'GBP' });
  }
  function trackSlot(raw){
    funnelPush({ event:'slot_select', service:FUNNEL_SERVICE, value:funnelValue(),
                 currency:'GBP', slot_date:String(raw).slice(0,10) });
  }
  function trackCheckout(){
    var v = funnelValue();
    funnelPush({ ecommerce:null });
    funnelPush({ event:'begin_checkout', service:FUNNEL_SERVICE, deposit:depositAmount(),
      ecommerce:{ currency:'GBP', value:v,
        items:[{ item_name:FUNNEL_LABEL, item_category:'Detailing', price:v, quantity:1 }] } });
  }
  function trackLead(method){
    funnelPush({ event:'generate_lead', service:FUNNEL_SERVICE, method:method });
  }

  var MONTHS = ['January','February','March','April','May','June',
                'July','August','September','October','November','December'];

  function loadDays(month){
    var type = bookingTypeId();
    if(!type) return;
    booking.month = month;

    var grid = root.querySelector('#odrBkDays');
    grid.innerHTML = '';
    root.querySelector('#odrBkTimesWrap').hidden = true;
    root.querySelector('#odrBkDaysLoad').hidden = false;
    bkErr.hidden = true;

    var y = +month.slice(0,4), m = +month.slice(5,7);
    root.querySelector('#odrBkMonthName').textContent = MONTHS[m-1] + ' ' + y;
    /* can't go back past this month */
    var now = new Date();
    root.querySelector('#odrBkPrev').disabled =
      (y < now.getFullYear()) || (y === now.getFullYear() && m <= now.getMonth() + 1);

    var p = addonParams(); p.type = type; p.month = month;
    askAcuity(p).then(function(days){
      root.querySelector('#odrBkDaysLoad').hidden = true;
      var free = {};
      (days || []).forEach(function(d){ free[d.date || d] = true; });
      drawMonth(y, m, free);
      if(!days || !days.length){
        /* nothing left this month — step forward rather than showing an
           empty grid and making them work it out */
        if(!booking.autoSkipped){
          booking.autoSkipped = true;
          shiftMonth(1);
          return;
        }
        bkFail('Nothing free in ' + MONTHS[m-1] + ' \u2014 try the next month.');
      }
    }).catch(function(){
      root.querySelector('#odrBkDaysLoad').hidden = true;
      bkFail('Couldn\u2019t reach the diary. Try again, or message us on WhatsApp.');
    });
  }

  function drawMonth(y, m, free){
    var grid = root.querySelector('#odrBkDays');
    grid.innerHTML = '';

    var first = new Date(y, m - 1, 1);
    var lead  = (first.getDay() + 6) % 7;          /* weeks start Monday */
    var count = new Date(y, m, 0).getDate();

    for(var i = 0; i < lead; i++){
      var blank = document.createElement('span');
      blank.className = 'bk-cell is-empty';
      grid.appendChild(blank);
    }

    for(var day = 1; day <= count; day++){
      var iso = y + '-' + ('0' + m).slice(-2) + '-' + ('0' + day).slice(-2);
      var b = document.createElement('button');
      b.type = 'button'; b.className = 'bk-cell'; b.textContent = day;
      b.setAttribute('aria-pressed','false');
      if(!free[iso]){
        b.disabled = true;
        b.setAttribute('aria-label', day + ' \u2014 fully booked');
      } else {
        b.setAttribute('data-iso', iso);
        (function(isoDay, btn){
          btn.addEventListener('click', function(){
            Array.prototype.forEach.call(grid.children, function(o){
              if(o.setAttribute) o.setAttribute('aria-pressed','false');
            });
            btn.setAttribute('aria-pressed','true');
            booking.day = isoDay; booking.time = null;
            bkPay.disabled = true; bkPay.textContent = 'Pay the deposit & book';
            root.querySelector('#odrBkChosen').hidden = true;
            loadTimes(isoDay);
          });
        })(iso, b);
      }
      grid.appendChild(b);
    }
  }

  var slotsByHour = {};

  function loadTimes(iso){
    var type = bookingTypeId();
    root.querySelector('#odrBkTimesWrap').hidden = false;
    root.querySelector('#odrBkTimesLoad').hidden = false;
    root.querySelector('#odrBkHours').innerHTML = '';
    root.querySelector('#odrBkMins').innerHTML  = '';

    var p = addonParams(); p.type = type; p.when = iso;
    askAcuity(p).then(function(times){
      root.querySelector('#odrBkTimesLoad').hidden = true;
      slotsByHour = {};
      (times || []).forEach(function(t){
        var raw = t.time || t;
        var hm  = String(raw).match(/T(\d{2}):(\d{2})/);
        if(!hm) return;
        (slotsByHour[hm[1]] = slotsByHour[hm[1]] || []).push({ mm:hm[2], raw:raw });
      });

      var hours = Object.keys(slotsByHour).sort();
      if(!hours.length){
        root.querySelector('#odrBkHours').innerHTML = '<p class="bk-load" style="padding:10px;">That day filled up.</p>';
        return;
      }
      hours.forEach(function(hh){
        var b = document.createElement('button');
        b.type = 'button'; b.className = 'bk-tick'; b.textContent = hh;
        b.setAttribute('aria-pressed','false');
        b.addEventListener('click', function(){
          Array.prototype.forEach.call(root.querySelector('#odrBkHours').children, function(o){
            if(o.setAttribute) o.setAttribute('aria-pressed','false');
          });
          b.setAttribute('aria-pressed','true');
          drawMinutes(hh);
        });
        root.querySelector('#odrBkHours').appendChild(b);
      });
      drawMinutes(hours[0]);
      root.querySelector('#odrBkHours').firstChild.setAttribute('aria-pressed','true');
    }).catch(function(){
      root.querySelector('#odrBkTimesLoad').hidden = true;
      bkFail('Couldn\u2019t load that day. Try another.');
    });
  }

  function drawMinutes(hh){
    var wrap = root.querySelector('#odrBkMins');
    wrap.innerHTML = '';
    (slotsByHour[hh] || []).forEach(function(slot){
      var b = document.createElement('button');
      b.type = 'button'; b.className = 'bk-tick'; b.textContent = slot.mm;
      b.setAttribute('aria-pressed','false');

      wrap.appendChild(b);
    });
  }
