<script>
(function(){
  var wrap = document.querySelector('.mps-wrap');
  if(!wrap) return;

  var bar      = wrap.querySelector('.mps-jump');
  var jumpbar  = wrap.querySelector('.mps-jumpbar');
  var arrowL   = wrap.querySelector('.mps-arrow-l');
  var arrowR   = wrap.querySelector('.mps-arrow-r');
  var switches = wrap.querySelectorAll('.mps-switch a');
  var garage   = wrap.querySelector('#garage-services');
  var io = null;

  /* ---- which half is showing ---- */
  function setMode(mode, focusTop){
    wrap.classList.remove('mps-m-mobile','mps-m-garage');
    wrap.classList.add(mode === 'garage' ? 'mps-m-garage' : 'mps-m-mobile');
    for(var i=0;i<switches.length;i++){
      switches[i].setAttribute('aria-current',
        switches[i].getAttribute('href') ===
        (mode === 'garage' ? '#garage-services' : '#mobile-services') ? 'true' : 'false');
    }
    if(bar) bar.scrollLeft = 0;
    watch();
    arrows();
    /* Squarespace measures this block's height once and keeps it. Poke it
       so it re-measures after the content gets shorter or taller. */
    reflow();
    if(focusTop && bar){
      var y = bar.getBoundingClientRect().top + window.pageYOffset - 8;
      if(window.pageYOffset > y) window.scrollTo({ top: y, behavior: 'smooth' });
    }
  }

  for(var i=0;i<switches.length;i++){
    (function(a){
      a.addEventListener('click', function(e){
        e.preventDefault();
        setMode(a.getAttribute('href') === '#garage-services' ? 'garage' : 'mobile', true);
      });
    })(switches[i]);
  }

  /* ---- horizontal scrolling for the pills on a desktop ----
     Touch users swipe. With a mouse there's no scrollbar to grab, so the
     arrows do it, and a vertical wheel over the bar scrolls it sideways. */
  function arrows(){
    if(!bar || !jumpbar || !arrowL || !arrowR) return;
    var over = bar.scrollWidth - bar.clientWidth > 2;
    jumpbar.classList.toggle('mps-ovf', over);
    arrowL.disabled = !over || bar.scrollLeft <= 1;
    arrowR.disabled = !over || bar.scrollLeft >= bar.scrollWidth - bar.clientWidth - 1;
  }

  function nudge(dir){
    if(!bar) return;
    bar.scrollBy({ left: dir * Math.max(140, bar.clientWidth * 0.7), behavior: 'smooth' });
  }

  if(arrowL) arrowL.addEventListener('click', function(){ nudge(-1); });
  if(arrowR) arrowR.addEventListener('click', function(){ nudge(1); });
  if(bar){
    bar.addEventListener('scroll', arrows, { passive: true });
    bar.addEventListener('wheel', function(e){
      if(Math.abs(e.deltaY) <= Math.abs(e.deltaX)) return;
      if(bar.scrollWidth - bar.clientWidth <= 2) return;
      e.preventDefault();
      bar.scrollLeft += e.deltaY;
    }, { passive: false });
  }
  window.addEventListener('resize', arrows);

  /* ---- make Squarespace re-measure the block height ---- */
  function reflow(){
    try{
      window.dispatchEvent(new Event('resize'));
      setTimeout(function(){ window.dispatchEvent(new Event('resize')); }, 120);
    }catch(e){}
  }

  /* ---- highlight the category you're reading, for the visible half only ---- */
  function watch(){
    if(!('IntersectionObserver' in window)) return;
    if(io) io.disconnect();

    var list = wrap.querySelector(
      wrap.classList.contains('mps-m-garage') ? '.mps-jump-garage' : '.mps-jump-mobile');
    if(!list) return;

    var links = Array.prototype.slice.call(list.querySelectorAll('a'));
    var secs  = links.map(function(a){ return wrap.querySelector(a.getAttribute('href')); });

    io = new IntersectionObserver(function(entries){
      entries.forEach(function(e){
        if(!e.isIntersecting) return;
        var i = secs.indexOf(e.target);
        if(i < 0) return;
        links.forEach(function(a){ a.classList.remove('mps-here'); });
        links[i].classList.add('mps-here');
        if(bar && bar.scrollWidth > bar.clientWidth){
          bar.scrollTo({ left: links[i].offsetLeft - 16, behavior: 'smooth' });
        }
      });
    }, { rootMargin: '-25% 0px -65% 0px', threshold: 0 });

    secs.forEach(function(s){ if(s) io.observe(s); });
  }

  /* ---- a link from elsewhere straight to a garage service opens that half ---- */
  function fromHash(){
    var h = window.location.hash;
    if(h && garage && (h === '#garage-services' || garage.querySelector(h))) return 'garage';
    return 'mobile';
  }

  setMode(fromHash(), false);
  arrows();
  setTimeout(arrows, 200);
  window.addEventListener('hashchange', function(){ setMode(fromHash(), false); });
})();
</script>
