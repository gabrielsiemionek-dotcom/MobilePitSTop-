<script>
(function(){
  if(window.mpsEmail) return;

  function enc(v){ return encodeURIComponent(v || ''); }

  function copyText(text, done){
    function legacy(){
      try{
        var ta = document.createElement('textarea');
        ta.value = text; ta.setAttribute('readonly', '');
        ta.style.position = 'fixed'; ta.style.opacity = '0';
        document.body.appendChild(ta); ta.select();
        var ok = document.execCommand('copy');
        document.body.removeChild(ta);
        done(ok);
      }catch(e){ done(false); }
    }
    if(navigator.clipboard && window.isSecureContext){
      navigator.clipboard.writeText(text).then(function(){ done(true); }, legacy);
    } else {
      legacy();
    }
  }

  /* to, subject, body, cc (optional), the button that was pressed */
  window.mpsEmail = function(to, subject, body, cc, button){
    /* 1. try the email app, exactly as before */
    var mailto = 'mailto:' + to + '?subject=' + enc(subject) +
                 (cc ? '&cc=' + enc(cc) : '') + '&body=' + enc(body);
    window.location.href = mailto;

    /* 2. and offer the browser routes underneath, in case nothing opened */
    var gmail = 'https://mail.google.com/mail/?view=cm&fs=1&to=' + enc(to) +
                '&su=' + enc(subject) + '&body=' + enc(body) + (cc ? '&cc=' + enc(cc) : '');
    var outlook = 'https://outlook.live.com/mail/0/deeplink/compose?to=' + enc(to) +
                  '&subject=' + enc(subject) + '&body=' + enc(body) + (cc ? '&cc=' + enc(cc) : '');

    var anchor = (button && button.parentNode) || null;
    if(!anchor || !anchor.parentNode) return;
    var box = anchor.parentNode.querySelector('.mps-mailfb');
    if(!box){
      box = document.createElement('div');
      box.className = 'mps-mailfb';
      box.setAttribute('role', 'status');
      anchor.parentNode.insertBefore(box, anchor.nextSibling);
    }
    box.innerHTML =
      '<p><strong>No email app opened?</strong> Send it another way &mdash; your message is already filled in.</p>' +
      '<div class="mps-mailfb-row">' +
        '<a href="' + gmail + '" target="_blank" rel="noopener">Open in Gmail</a>' +
        '<a href="' + outlook + '" target="_blank" rel="noopener">Open in Outlook</a>' +
        '<button type="button" data-mps-copy>Copy message</button>' +
      '</div>' +
      '<p class="mps-mailfb-addr">Or paste it into an email to <a href="mailto:' + to + '">' + to + '</a></p>';

    var copyBtn = box.querySelector('[data-mps-copy]');
    copyBtn.addEventListener('click', function(){
      copyText('Subject: ' + subject + '\n\n' + body, function(ok){
        copyBtn.textContent = ok ? 'Copied ✓' : 'Copy failed — select the message above';
        setTimeout(function(){ copyBtn.textContent = 'Copy message'; }, 2500);
      });
    });

    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({ event:'pf_email_fallback_shown' });
  };
})();
</script>

<script>
(function(){
  var root = document.getElementById('wtRoot');
  if(!root) return;

  /* ====================================================================
     GO LIVE SWITCH
     While this is false the block hides itself and shows a short holding
     note instead, so the page can be published before the service is.
     Flip it to true when it's approved — nothing else to change.
     ==================================================================== */
  var APPROVED = true;

  var SERVICE   = "window tinting";
  var WHATSAPP_NUMBER = '447563718029';   /* the tint fitter — tint requests go straight to them */
  var EMAIL_TO        = 'mobilepitstopvalet@gmail.com';

  /* Second pair of eyes on this service.
     PARTNER_EMAIL is CC'd automatically on the email route.
     PARTNER_WHATSAPP gets a "send a copy" button after the first send —
     WhatsApp has no way to start a GROUP chat with a message already in
     it, so a copy to a second number is the closest thing that works
     every time. (See the note at the top of this file.) */
  var PARTNER_EMAIL    = "";
  var PARTNER_WHATSAPP = "447592196929";   /* MobilePitStop gets the copy */

  var rqName = root.querySelector('#wtRqName');
  var rqCar  = root.querySelector('#wtRqCar');
  var rqYear = root.querySelector('#wtRqYear');
  var rqDate = root.querySelector('#wtRqDate');
  var rqMsg  = root.querySelector('#wtRqMsg');
  var rqErr  = root.querySelector('#wtRqErr');
  var rqDirty = false;

  if(rqDate){
    try{
      var tm = new Date(); tm.setDate(tm.getDate() + 1);
      rqDate.min = tm.toISOString().slice(0, 10);
    }catch(e){}
  }
  function niceDate(iso){
    if(!iso) return '';
    try{
      return new Date(iso + 'T12:00:00').toLocaleDateString('en-GB',
        { weekday:'short', day:'numeric', month:'short', year:'numeric' });
    }catch(e){ return iso; }
  }

  /* ======================================================================
     FUNNEL TRACKING
     dataLayer pushes only. GTM (GTM-MMHW32R2) picks them up and sends
     them to GA4, so there is no gtag/GA code here. Values are numbers in
     pounds, after any promo. Never put a name, email, phone, address,
     postcode or reg into these.
       quote_start     first choice made           (once per page load)
       generate_lead   a WhatsApp or email request sent
     ====================================================================== */
  var FUNNEL_SERVICE = 'window_tinting';
  var funnelSent;
  function funnelPush(o){
    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push(o);
  }
  function funnelOnce(key, o){
    funnelSent = funnelSent || {};
    if(funnelSent[key]) return;
    funnelSent[key] = true;
    funnelPush(o);
  }
  function trackQuoteStart(){
    funnelOnce('start', { event:'quote_start', service:FUNNEL_SERVICE });
  }
  function trackLead(method){
    funnelPush({ event:'generate_lead', service:FUNNEL_SERVICE, method:method });
  }

  /* chips: multi-select, and every choice ends up in the message */
  var chips = Array.prototype.slice.call(root.querySelectorAll('.qopt'));
  function picked(groupId){
    return chips.filter(function(c){
      return c.getAttribute('data-group') === groupId && c.getAttribute('aria-pressed') === 'true';
    }).map(function(c){ return c.getAttribute('data-value'); });
  }
  chips.forEach(function(c){
    c.addEventListener('click', function(){
      trackQuoteStart();
      c.setAttribute('aria-pressed', c.getAttribute('aria-pressed') === 'true' ? 'false' : 'true');
      refreshMessage();
    });
  });

  var GROUPS = [{"id": "windows", "label": "Windows"}, {"id": "film", "label": "Film"}, {"id": "shade", "label": "Shade"}];

  function requestMessage(){
    var L = [];
    L.push('Hi MobilePitStop, I\u2019d like a quote for ' + SERVICE + '.');
    L.push('');
    if(rqCar && rqCar.value.trim()){
      L.push('Vehicle: ' + rqCar.value.trim() +
             (rqYear && rqYear.value.trim() ? ' (' + rqYear.value.trim() + ')' : ''));
    }
    GROUPS.forEach(function(g){
      var vals = picked(g.id);
      if(vals.length) L.push(g.label + ': ' + vals.join(', '));
    });
    if(rqName && rqName.value.trim()) L.push('Name: ' + rqName.value.trim());
    if(rqDate && rqDate.value)        L.push('Preferred week: ' + niceDate(rqDate.value));
    return L.join('\n');
  }
  function refreshMessage(){ if(rqMsg && !rqDirty) rqMsg.value = requestMessage(); }
  function showErr(t){ if(!rqErr) return; rqErr.hidden = !t; rqErr.textContent = t || ''; }
  function valid(){
    if(!rqName || !rqName.value.trim()){ showErr('Add your name so we know who we\u2019re talking to.'); if(rqName) rqName.focus(); return false; }
    if(!rqCar || !rqCar.value.trim()){ showErr('Tell us the make and model so we can price it.'); if(rqCar) rqCar.focus(); return false; }
    if(!rqYear || !rqYear.value.trim()){ showErr('Add the year — trim and panel shapes change between facelifts.'); if(rqYear) rqYear.focus(); return false; }
    showErr('');
    return true;
  }
  function track(via){
    if(via === 'whatsapp' || via === 'email') trackLead(via);
    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({ event:'pf_request_click', component:'quote_request',
                            service:"window_tinting", via:via });
  }

  [rqName, rqCar, rqYear, rqDate].forEach(function(el){
    if(!el) return;
    el.addEventListener('input',  function(){ showErr(''); refreshMessage(); });
    el.addEventListener('change', refreshMessage);
  });
  if(rqMsg) rqMsg.addEventListener('input', function(){ rqDirty = true; });

  function sendWhatsApp(number, text){
    var url = 'https://wa.me/' + number + '?text=' + encodeURIComponent(text);
    var w = window.open(url, '_blank', 'noopener');
    if(!w) window.location.href = url;
  }

  root.querySelector('#wtRqWhatsApp').addEventListener('click', function(){
    if(!valid()) return;
    var text = rqMsg && rqMsg.value.trim() ? rqMsg.value : requestMessage();
    track('whatsapp');
    sendWhatsApp(WHATSAPP_NUMBER, text);
    /* offer the copy AFTER the first send, so one tap isn't two popups */
    var also = root.querySelector('#wtRqAlso');
    if(PARTNER_WHATSAPP && also) also.hidden = false;
  });

  var alsoBtn = root.querySelector('#wtRqAlsoBtn');
  if(alsoBtn){
    alsoBtn.addEventListener('click', function(){
      var text = rqMsg && rqMsg.value.trim() ? rqMsg.value : requestMessage();
      track('whatsapp_partner');
      sendWhatsApp(PARTNER_WHATSAPP, text);
    });
  }

  root.querySelector('#wtRqEmail').addEventListener('click', function(){
    if(!valid()) return;
    var text = rqMsg && rqMsg.value.trim() ? rqMsg.value : requestMessage();
    track('email');
    var subject = "Window tint quote request" + (rqCar && rqCar.value.trim() ? ' \u2014 ' + rqCar.value.trim() : '');
    window.mpsEmail(EMAIL_TO, subject, text, PARTNER_EMAIL, root.querySelector('#wtRqEmail'));
  });

  var learn = document.getElementById('wtLearnMoreBtn');
  if(learn){
    learn.addEventListener('click', function(){
      var el = document.getElementById("learnMoreWindowTint");
      if(el) el.scrollIntoView({ behavior:'smooth', block:'start' });
    });
  }

  if(!APPROVED){
    root.setAttribute('hidden', '');
    var holder = document.createElement('div');
    holder.className = 'pf';
    holder.innerHTML =
      '<div class="topbar"><div><h1>Window Tinting</h1>' +
      '<h3>Coming soon.</h3></div></div>' +
      '<div class="info-bar">We’re not taking window tinting bookings through the website just yet. ' +
      'Call or message <a href="tel:07592196929">07592 196929</a> and we’ll talk it through.</div>';
    root.parentNode.insertBefore(holder, root);
    return;
  }

  refreshMessage();
})();
</script>
