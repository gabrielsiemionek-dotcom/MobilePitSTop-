<script>
(function(){
  var root = document.getElementById('mpfFinder');
  if(!root) return;
  var step1 = root.querySelector('#mpfStep1');
  var lastOpener = null;

  function show(step){
    Array.prototype.forEach.call(root.querySelectorAll('.mpf-step'), function(s){
      s.hidden = (s !== step);
    });
  }

  /* "Deep Clean" / "Professional Detail": open the second question */
  Array.prototype.forEach.call(root.querySelectorAll('[data-next]'), function(a){
    a.setAttribute('role', 'button');
    a.setAttribute('aria-controls', a.getAttribute('data-next'));
    a.addEventListener('click', function(e){
      e.preventDefault();
      var step = root.querySelector('#' + a.getAttribute('data-next'));
      lastOpener = a;
      show(step);
      step.querySelector('.mpf-q').focus({ preventScroll: true });
      var top = root.getBoundingClientRect().top;
      if(top < 0) root.scrollIntoView({ block: 'start' });
    });
    a.addEventListener('keydown', function(e){
      if(e.key === ' ' || e.key === 'Spacebar'){ e.preventDefault(); a.click(); }
    });
  });

  Array.prototype.forEach.call(root.querySelectorAll('.mpf-back'), function(b){
    b.addEventListener('click', function(){
      show(step1);
      if(lastOpener) lastOpener.focus();
    });
  });

  /* a service was chosen */
  Array.prototype.forEach.call(root.querySelectorAll('a[data-svc]'), function(a){
    a.addEventListener('click', function(){
      window.dataLayer = window.dataLayer || [];
      window.dataLayer.push({ event: 'service_finder_click',
        service: a.getAttribute('data-svc'), choice: a.getAttribute('data-route') });
    });
  });
})();
</script>

<script>
(function(){
  var ELFSIGHT_APP = 'elfsight-app-b97c8bc8-5963-4b15-a78b-2da5362c47dc';
  var ELFSIGHT_JS  = 'https://elfsightcdn.com/platform.js';

  /* load platform.js once per page, even if another block already asked for it */
  function loadPlatform(){
    if(document.querySelector('script[src="' + ELFSIGHT_JS + '"]')) return;
    var sc = document.createElement('script');
    sc.src = ELFSIGHT_JS;
    sc.async = true;
    document.body.appendChild(sc);
  }

  function setup(slot){
    if(slot.getAttribute('data-mps-rvm-ready')) return;
    slot.setAttribute('data-mps-rvm-ready', '1');
    var wait = slot.querySelector('[data-mps-rvm-wait]');

    var loaded = false;
    function loadReviews(){
      if(loaded) return;
      loaded = true;

      var app = document.createElement('div');
      app.className = ELFSIGHT_APP;
      app.setAttribute('data-elfsight-app-lazy', '');
      slot.appendChild(app);
      loadPlatform();

      setTimeout(function(){ if(wait) wait.style.display = 'none'; }, 1500);
    }

    if('IntersectionObserver' in window){
      var io = new IntersectionObserver(function(entries){
        entries.forEach(function(e){
          if(e.isIntersecting){ loadReviews(); io.disconnect(); }
        });
      }, { rootMargin: '400px 0px' });
      io.observe(slot);
    } else {
      window.addEventListener('scroll', loadReviews, { once:true, passive:true });
      setTimeout(loadReviews, 4000);
    }
  }

  var slots = document.querySelectorAll('[data-mps-rvm-slot]');
  for(var i = 0; i < slots.length; i++) setup(slots[i]);
})();
</script>
